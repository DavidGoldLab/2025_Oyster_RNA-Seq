# Clean data with BBDuk.sh in BBTools (v.39.01)
################################################

module load bbmap # v.39.01
module load openjdk # v.16.0.2

# Create "polyA.fa" file
echo ">polyA
AAAAAAAAAAAAAAAAAA" > polyA.fa

# Get "truseq_rna.fa.gz" dataset
wget https://github.com/ben-laufer/Tag-seq/raw/fab6d6afd181dad5dfd08bb5e641c2ed1ca26802/data/truseq_rna.fa.gz

for i in *.fastq ; do \
 bbduk.sh in=$i out=${i%.fastq}.clean.fq ref=polyA.fa,truseq_rna.fa.gz \
 k=13 ktrim=r useshortkmers=t mink=5 qtrim=r trimq=10 minlength=20;\
done

# use FastQC (v.0.11.9) to check the cleaned data
module load fastqc # v.0.11.9
mkdir qualitycheck
for i in *clean.fq ; do \
 fastqc --outdir qualitycheck --format fastq \
 --threads 16 $i;\
done

rm *fastq

###################################
# 1. Map reads to genome with STAR
###################################
###################################

cd ../1_Mapping

module load star # v.2.7.6a

# Make STAR genome index
STAR --runThreadN 16 \
--runMode genomeGenerate \
--genomeDir ./genome \
--genomeFastaFiles ../0_Data/GCF_963853765.1_xbMagGiga1.1_genomic.fna \
--sjdbGTFfile ../0_Data/genomic.gff \
--sjdbOverhang 100

# Map reads to genome with STAR
for i in ../0_Data/*.fq; do
  filename=$(basename "$i")
  STAR --runThreadN 16 --genomeDir ./genome --readFilesIn ./$i \
  --outFilterType BySJout --outFilterMultimapNmax 20 --alignSJoverhangMin 8 --alignSJDBoverhangMin 1 \
  --outFilterMismatchNmax 999 --outFilterMismatchNoverLmax 0.1 --alignIntronMin 20 \
  --alignIntronMax 1000000 --alignMatesGapMax 1000000 --outSAMattributes NH HI NM MD \
  --outSAMtype BAM SortedByCoordinate --outFileNamePrefix ${filename%.clean.fq} ;\
done
